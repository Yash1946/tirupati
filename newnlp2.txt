# =========================================================
# Step 6: Word2Vec Embeddings (Skip-Gram using PyTorch)
# =========================================================

# Tokenize and clean documents for Word2Vec

stop_words = set(stopwords.words('english'))

tokenized_docs = []

all_words = []

for doc in documents:

    tokens = word_tokenize(doc.lower())

    tokens = [
        t for t in tokens
        if t.isalpha() and t not in stop_words
    ]

    tokenized_docs.append(tokens)

    all_words.extend(tokens)

print('\nTokenized Documents (after cleaning):')

for i, tokens in enumerate(tokenized_docs, 1):

    print(f'Doc {i}: {tokens}')


# ---------------------------------------------------------
# Build vocabulary
# ---------------------------------------------------------

word_counts = Counter(all_words)

vocab = sorted(word_counts.keys())

word_to_idx = {
    w: i for i, w in enumerate(vocab)
}

idx_to_word = {
    i: w for w, i in word_to_idx.items()
}

vocab_size = len(vocab)

print(f'\nVocabulary size: {vocab_size}')

print(f'Vocabulary: {vocab}')


# ---------------------------------------------------------
# Generate Skip-Gram training pairs
# ---------------------------------------------------------

# For each word, pair it with neighboring words within a context window

window_size = 2

training_pairs = []

for doc_tokens in tokenized_docs:

    for i, target in enumerate(doc_tokens):

        # Get context words within the window
        start = max(0, i - window_size)

        end = min(len(doc_tokens), i + window_size + 1)

        for j in range(start, end):

            if i != j:  # Skip the target word itself

                context = doc_tokens[j]

                training_pairs.append(
                    (
                        word_to_idx[target],
                        word_to_idx[context]
                    )
                )

print(f'\nTotal training pairs: {len(training_pairs)}')

print('\nSample pairs (first 10):')

for t, c in training_pairs[:10]:

    print(
        f'Target: {idx_to_word[t]:<15} '
        f'-> Context: {idx_to_word[c]}'
    )


# =========================================================
# Define Skip-Gram Word2Vec model
# =========================================================

class SkipGramWord2Vec(nn.Module):

    def __init__(self, vocab_size, embedding_dim):

        super(SkipGramWord2Vec, self).__init__()

        # Target word embeddings
        self.target_embeddings = nn.Embedding(
            vocab_size,
            embedding_dim
        )

        # Context word embeddings
        self.context_embeddings = nn.Embedding(
            vocab_size,
            embedding_dim
        )

    def forward(self, target, context):

        target_emb = self.target_embeddings(target)

        context_emb = self.context_embeddings(context)

        # Dot product for similarity score
        scores = torch.sum(
            target_emb * context_emb,
            dim=1
        )

        return scores

    def get_embedding(self, word_idx):

        return self.target_embeddings(
            torch.tensor([word_idx])
        ).detach().numpy()[0]


# ---------------------------------------------------------
# Initialize model
# ---------------------------------------------------------

EMBEDDING_DIM = 30

w2v_model = SkipGramWord2Vec(
    vocab_size,
    EMBEDDING_DIM
)

print('\nSkip-Gram Word2Vec Model:')

print(f'Vocabulary size: {vocab_size}')

print(f'Embedding dim: {EMBEDDING_DIM}')

print(
    f'Parameters: '
    f'{sum(p.numel() for p in w2v_model.parameters()):,}'
)


# =========================================================
# Train the Word2Vec model
# =========================================================

optimizer = optim.Adam(
    w2v_model.parameters(),
    lr=0.01
)

criterion = nn.BCEWithLogitsLoss()

# Prepare training data (positive pairs)

targets = torch.tensor(
    [p[0] for p in training_pairs]
)

contexts = torch.tensor(
    [p[1] for p in training_pairs]
)

labels = torch.ones(len(training_pairs))

# Add negative samples (random word pairs)

num_neg = len(training_pairs)

neg_targets = torch.randint(
    0,
    vocab_size,
    (num_neg,)
)

neg_contexts = torch.randint(
    0,
    vocab_size,
    (num_neg,)
)

neg_labels = torch.zeros(num_neg)

# Combine positive and negative

all_targets = torch.cat([
    targets,
    neg_targets
])

all_contexts = torch.cat([
    contexts,
    neg_contexts
])

all_labels = torch.cat([
    labels,
    neg_labels
])

# Training loop

print('\nTraining Word2Vec (Skip-Gram)...')

print('-' * 40)

num_epochs = 100

for epoch in range(num_epochs):

    w2v_model.train()

    optimizer.zero_grad()

    scores = w2v_model(
        all_targets,
        all_contexts
    )

    loss = criterion(scores, all_labels)

    loss.backward()

    optimizer.step()

    if (epoch + 1) % 20 == 0:

        print(
            f'Epoch {epoch+1:3d}/{num_epochs} '
            f'| Loss: {loss.item():.4f}'
        )

print('\nTraining complete!')

# =========================================================
# Find similar words using cosine similarity
# =========================================================

import torch.nn.functional as F

def find_similar(word, top_n=3):

    if word not in word_to_idx:

        return []

    word_vec = torch.tensor(
        w2v_model.get_embedding(
            word_to_idx[word]
        )
    ).unsqueeze(0)

    similarities = []

    for other_word, idx in word_to_idx.items():

        if other_word == word:
            continue

        other_vec = torch.tensor(
            w2v_model.get_embedding(idx)
        ).unsqueeze(0)

        sim = F.cosine_similarity(
            word_vec,
            other_vec
        ).item()

        similarities.append((other_word, sim))

    similarities.sort(
        key=lambda x: x[1],
        reverse=True
    )

    return similarities[:top_n]


print('\nMost Similar Words (cosine similarity):')

print('-' * 50)

for word in ['language', 'learning', 'text']:

    if word in word_to_idx:

        similar = find_similar(word, 3)

        print(f'\n{word}:')

        for sim_word, score in similar:

            print(
                f'  -> {sim_word} '
                f'(similarity: {score:.4f})'
            )
