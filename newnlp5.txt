# Define vowels and consonants

VOWELS = set('aeiou')


def ends_with_consonant_y(word):
    """
    Check if word ends with consonant followed by y.
    """

    return (
        len(word) > 2 and
        word[-1] == 'y' and
        word[-2] not in VOWELS
    )


def ends_with_sibilant(word):
    """
    Check if word ends with s, sh, ch, x, or z.
    """

    return (
        word.endswith('s') or
        word.endswith('sh') or
        word.endswith('ch') or
        word.endswith('x') or
        word.endswith('z')
    )


def apply_add_delete(word, delete, add):
    """
    Apply add-delete rule:
    remove 'delete' from end,
    then append 'add'.
    """

    if delete and word.endswith(delete):

        word = word[:-len(delete)]

    return word + add


print('Helper functions defined successfully!')


# =========================================================
# Generate Word Forms
# =========================================================

def generate_word_forms(base_word, word_type='noun'):
    """
    Generate different morphological forms
    of a word using Add-Delete table approach.
    """

    forms = {'base': base_word}

    # -----------------------------------------------------
    # Verb Rules
    # -----------------------------------------------------

    if word_type == 'verb':

        # Present participle (-ing)

        if base_word.endswith('e'):

            forms['present_participle'] = apply_add_delete(
                base_word,
                'e',
                'ing'
            )

        else:

            forms['present_participle'] = apply_add_delete(
                base_word,
                '',
                'ing'
            )

        # Past tense (-ed / -d)

        if base_word.endswith('e'):

            forms['past_tense'] = apply_add_delete(
                base_word,
                '',
                'd'
            )

        else:

            forms['past_tense'] = apply_add_delete(
                base_word,
                '',
                'ed'
            )

        # Third person singular (-s / -es)

        if ends_with_sibilant(base_word):

            forms['third_person'] = apply_add_delete(
                base_word,
                '',
                'es'
            )

        else:

            forms['third_person'] = apply_add_delete(
                base_word,
                '',
                's'
            )

    # -----------------------------------------------------
    # Noun Rules
    # -----------------------------------------------------

    elif word_type == 'noun':

        # Plural

        if ends_with_consonant_y(base_word):

            forms['plural'] = apply_add_delete(
                base_word,
                'y',
                'ies'
            )

        elif ends_with_sibilant(base_word):

            forms['plural'] = apply_add_delete(
                base_word,
                '',
                'es'
            )

        else:

            forms['plural'] = apply_add_delete(
                base_word,
                '',
                's'
            )

    # -----------------------------------------------------
    # Adjective Rules
    # -----------------------------------------------------

    elif word_type == 'adjective':

        # Negation
        forms['negation'] = 'un' + base_word

        # Noun form (-ness)
        forms['noun_form'] = apply_add_delete(
            base_word,
            '',
            'ness'
        )

        # Adverb form (-ly)
        forms['adverb_form'] = apply_add_delete(
            base_word,
            '',
            'ly'
        )

    return forms

# Step 5: Morphological Analysis (Reverse Process)
# =========================================================

def analyze_morphology(word):
    """
    Analyze a word and attempt to identify
    its base form and the morphological rule applied.
    """

    analyses = []

    # -----------------------------------------------------
    # Check for prefixes
    # -----------------------------------------------------

    if word.startswith('un'):

        analyses.append({
            'word': word,
            'base': word[2:],
            'operation': 'Prefix: un- (negation)',
            'delete': '--',
            'add': 'un-'
        })

    # -----------------------------------------------------
    # Check suffix rules
    # -----------------------------------------------------

    suffix_rules = [

        ('ation', 'e', 'Suffix: -ation (noun from verb)'),

        ('iness', 'y', 'Suffix: -ness (noun from adj, y->iness)'),

        ('ness', '', 'Suffix: -ness (noun from adjective)'),

        ('ies', 'y', 'Suffix: -ies (plural, y->ies)'),

        ('ing', '', 'Suffix: -ing (present participle)'),

        ('ful', '', 'Suffix: -ful (adjective from noun)'),

        ('less', '', 'Suffix: -less (adjective from noun)'),

        ('ed', '', 'Suffix: -ed (past tense)'),

        ('ly', '', 'Suffix: -ly (adverb from adjective)'),

        ('es', '', 'Suffix: -es (plural/3rd person)'),

        ('s', '', 'Suffix: -s (plural/3rd person)')
    ]

    for suffix, restore, description in suffix_rules:

        if word.endswith(suffix):

            base = word[:-len(suffix)] + restore

            analyses.append({
                'word': word,
                'base': base,
                'operation': description,
                'delete': suffix,
                'add': restore or '--'
            })

    return analyses
